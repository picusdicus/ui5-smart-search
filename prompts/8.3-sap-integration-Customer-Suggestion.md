Important: use ai-agent for this task

Read CLAUDE.md first, then implement the suggestCustomer
action. This suggests form fields for a NEW Business Partner
based on a similar existing one found via vector search.

This is a SUGGESTION feature, not a creation feature.

Dependencies:
- srv/lib/s4-client.js (fetchBusinessPartnerById)
- srv/lib/embeddings.js (embedText)
- srv/lib/hana-vector.js (vectorSearch)
- srv/lib/ollama-client.js (askLlama)

─────────────────────────────────────────────────
1. UPDATE srv/search-service.cds
─────────────────────────────────────────────────
Add action:

action suggestCustomer(query: String) returns {
  suggestedFields  : String;   // JSON: name, language, grouping
  similarBPId      : String;
  similarBPName    : String;
  duplicateWarning : Boolean;
};

─────────────────────────────────────────────────
2. UPDATE srv/search-service.js
─────────────────────────────────────────────────
Add handler:

on('suggestCustomer') {

  // STEP 1: vector search
  const vector = await embedText(query)
  const similar = await vectorSearch('Customers', vector, 1)

  if (!similar.length) return {
    suggestedFields: JSON.stringify({
      name: '', language: 'EN', grouping: 'BP01'
    }),
    similarBPId: '',
    similarBPName: '',
    duplicateWarning: false
  }

  const topMatch = similar[0]
  const isDuplicate = topMatch.score > 0.95

  // STEP 2: fetch reference BP from API Hub
  const bp = await fetchBusinessPartnerById(topMatch.ID)

  // STEP 3: ask Llama for name + language ONLY
  // grouping always comes from reference BP, never Llama
  const llamaPrompt = 
    `You are an SAP form assistant.
     Suggest a company name for this description:
     "${query}"
     Reference BP language: ${bp.language}
     Rules:
     - name: realistic company name inspired by 
       the description, NOT the description itself
     - language: uppercase ISO 639-1 code 
       e.g. EN DE FR ES, never country codes like US
     Return ONLY this JSON, no markdown, no extra fields:
     {"name":"...","language":"..."}`

  const { answer } = await askLlama(llamaPrompt, [], 'Customers')

  // STEP 4: clean markdown Llama might add
  const cleanJSON = answer
    .replace(/```json/g, '')
    .replace(/```/g, '')
    .trim()

  const parsed = JSON.parse(cleanJSON)

  // STEP 5: build final fields
  // always override grouping from reference BP
  const suggested = {
    name:     parsed.name,
    language: (parsed.language || 'EN').toUpperCase(),
    grouping: bp.grouping || 'BP01'
  }

  return {
    suggestedFields:  JSON.stringify(suggested),
    similarBPId:      topMatch.ID,
    similarBPName:    topMatch.name,
    duplicateWarning: isDuplicate
  }
}
