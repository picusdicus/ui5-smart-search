Create srv/lib/s4-client.js to connect to SAP API Hub.

Base URL: process.env.API_HUB_BASE_URL
Auth header: APIKey: process.env.API_HUB_KEY
Response format: OData V2 → parse d.results

Implement 3 functions:

fetchBusinessPartners(top=100)
  GET A_BusinessPartner
  $filter=BusinessPartnerCategory eq '2' 
    and Customer ne ''
    and IsMarkedForArchiving eq false
  $format=json
  Return array of: {
    id: BusinessPartner,
    name: BusinessPartnerFullName || OrganizationBPName1,
    language: CorrespondenceLanguage,
    lastChanged: LastChangeDate,
    isBlocked: BusinessPartnerIsBlocked
  }

fetchBusinessPartnerById(bpId)
  GET A_BusinessPartner('${bpId}')
  $format=json
  Return same mapped object

createBusinessPartner(data)
  POST A_BusinessPartner
  Body: {
    BusinessPartnerCategory: "2",
    BusinessPartnerGrouping: data.grouping || "BP01",
    OrganizationBPName1: data.name,
    CorrespondenceLanguage: data.language || "EN",
    BusinessPartnerIsBlocked: false
  }
  Return { id: result.BusinessPartner, success: true }
  On error: throw with error.message.value from response
